const API_URL = "https://dhofarcare-api.sharooqalmashani1.workers.dev"; const bookingForm = document.getElementById("booking-form"); const successMessage = document.getElementById("success-message"); bookingForm.addEventListener("submit", async function (event) { event.preventDefault(); const button = bookingForm.querySelector('button[type="submit"]'); const formData = new FormData(bookingForm); const booking = { name: String(formData.get("name") || "").trim(), email: String(formData.get("email") || "").trim(), phone: String(formData.get("phone") || "").trim(), area: String(formData.get("area") || ""), service: String(formData.get("service") || ""), datetime: String(formData.get("datetime") || "") }; button.disabled = true; button.textContent = "Sending request..."; successMessage.classList.add("hidden"); try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ type: "booking", booking: booking }) }); const result = await response.json(); if (!response.ok || !result.ok) { throw new Error(result.error || "Booking failed"); } successMessage.textContent = "✅ Request sent successfully! Reference: " + (result.reference || "N/A"); successMessage.classList.remove("hidden"); bookingForm.reset(); } catch (error) { console.error("Booking error:", error); successMessage.textContent = "❌ Could not send booking. Please try again."; successMessage.classList.remove("hidden"); } finally { button.disabled = false; button.textContent = "Submit Request"; } });
